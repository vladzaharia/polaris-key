Package deliverables docs
.pkey/release
deliverables:
  npm.sourcekit:
    kind: package
    ecosystem: npm # npm | pypi | swift | maven | oci | godot
    name: "@djdl/source-kit"
    artifacts:
      tarball: { match: "dist/npm/djdl-source-kit-*.tgz" }
pkey release publish --deliverable npm.sourcekit --dir dist/npm
