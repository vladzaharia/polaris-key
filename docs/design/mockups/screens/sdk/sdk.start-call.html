Node · main.ts
import { PolarisKeyClient }
  from "@polaris-key/node";

const pk = await PolarisKeyClient.fromConfig();
await pk.boot();

if (!(await pk.isLicensed())) {
  await showSignIn(pk);
}
React · main.tsx

pkey sdk add

Outside DJDL’s repository, add --product djdl
